def until(read):

def peer_request(payload):

def keys(codes,overrides=None,flags=0):
 events=[]
 for index,code in enumerate(codes):
  for down in (True,False):
   row={'type':'key_down' if down else 'key_up','keycode':code,'keyboard_type':0,'repeat':0,'flags':flags,'at_us':index*80000+(0 if down else 40000)}
   if overrides is not None:row['unicode']=overrides[index]
   events.append(row)
 return events
